# Banter Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Residents trash-talk at a table with the whole picture (players, turn, standing, the last 10 moves narrated, the last 10 chat lines), and answer to their name anywhere in a lobby or group line.

**Architecture:** Games narrate their own moves through a new optional `TurnRules.narrate`; the server keeps each live match's last 20 narrated entries in memory and sends them as `MatchEnvelope.recent`. The bots turn an envelope into a table context (players, turn, standing, moves) that `buildUserPrompt` lays out like the Foreman's playground. A per-resident `answersTo` pattern drives name-anywhere triggers in the lobby watcher, groups and at the table.

**Tech Stack:** Bun + TypeScript, `bun:test`, Hono server (`apps/server`), bots (`apps/bots`), game packages (`games/*`, `packages/game-api`, `packages/shared`).

**Spec:** `docs/superpowers/specs/2026-10-02-banter-design.md`

## Global Constraints

- The log lives in memory only: no database table, no migration, nothing written to `.data`.
- `.data/llm.json` is the Foreman's; never edit it (reading it for the live smell test is fine).
- Keep 20 entries per match (`RECENT_MOVES = 20`); bots show the last 10 moves and the last 10 chat lines.
- Table-talk instruction, word for word: `You are in the middle of a match at this table, and @X just said something. Answer their message in character — talk trash, name names when it stings. One line.`
- A narration or context failure must never break a move or a reply: fall back, never throw.
- House-bot guards, cooldowns (20 s per channel) and DM/group hourly caps stay exactly as they are.
- Run tests in the foreground; never background a verify run.
- Every commit message ends with `Co-Authored-By: Claude Opus 5.5 (1M context) <noreply@anthropic.com>`.

## Review Focus

- A game module whose `narrate` throws (or a game with no `narrate`) — the move still lands, logged as `plays <describe>`. (Task 2 test.)
- A `{seat}` placeholder naming a seat that isn't in the match — left as written, not crashed. (Task 2 test.)
- A table moment where fetching the envelope fails (match reaped, network blip) — the bot still speaks, without the context. (Task 3: `speak` catches; covered by the hello-on-time test staying green.)
- Names inside other names: `@OLD_IRON-2`, `OLD_IRONX`, `sprockets`, lowercase "game over" — no trigger. (Task 4 test.)
- A line that both @mentions the bot and names it in the lobby — exactly one reply, not two. (Task 4 test.)

```ts
/** The move for the recent-moves log: the chain's size, whose cells it took, who it knocked out. */
function narrate(before: CMState, after: CMState, seat: SeatId, command: unknown): string {
  const cell = String((command as CMCommand).cell);
  const waves = after.lastMove?.waves.length ?? 0;
  const parts: string[] = [];
  if (waves >= 3) parts.push('big chain!');
  else if (waves > 0) parts.push('small chain');
  const taken = before.seats
    .filter((x) => x !== seat)
    .map((x) => ({ x, n: after.cells.filter((c, i) => c.owner === seat && before.cells[i].owner === x).length }))
    .filter((t) => t.n > 0);
  if (taken.length > 0) parts.push(`takes ${taken.map((t, k) => (k === 0 ? `${t.n} of {${t.x}}'s cells` : `${t.n} of {${t.x}}'s`)).join(' and ')}`);
  for (const x of after.eliminated) if (x !== seat && !before.eliminated.includes(x)) parts.push(`knocks {${x}} out`);
  return parts.length > 0 ? `places at ${cell} — ${parts.join(', ')}` : `places at ${cell}`;
}
```

```ts
  /** Logs a line for the recent-moves list; `{seat}` placeholders become that seat's handle. */
  private record(m: LiveMatch, seat: SeatId | null, text: string, number?: number): void {
    const handleOf = (s: string) => m.seats.find((x) => x.seat === s)?.identity.handle;
    m.logged += 1;
    m.recent.push({
      number: number ?? m.logged,
      seat,
      handle: seat ? (handleOf(seat) ?? null) : null,
      text: text.replace(/\{([A-Za-z0-9_-]+)\}/g, (whole, s: string) => handleOf(s) ?? whole),
    });
    if (m.recent.length > RECENT_MOVES) m.recent.shift();
  }

  /** The game's own sentence for a move, or "plays <describe>"; a broken narrator never breaks a move. */
  private narrateMove(m: LiveMatch, before: unknown, after: unknown, seat: SeatId, command: unknown): string {
    const turn = m.module.turn!;
    try {
      if (turn.narrate) return turn.narrate(before, after, seat, command);
    } catch (e) {
      console.error(`[games] narrating a move in ${m.id} failed`, e);
    }
    return `plays ${turn.describe ? turn.describe(command) : JSON.stringify(command)}`;
  }
```

```ts
import type { MatchEnvelope } from '@cs/shared/protocol';

/** What a resident is told about the table it's sitting at, beside the chat. */
export interface TableContext {
  /** "@Frosty (human), @Fable (machine), @OLD_IRON (you)" */
  players: string;
  /** "Turn 36 — @Fable to move", "Round 2", "The match is over" */
  turn: string;
  /** "Fable leads with 14 cells, you hold 11, Frosty is down to 5"; null for a game we can't score. */
  standing: string | null;
  /** The last moves, oldest first: "28. OLD_IRON places at b4 — small chain, takes 2 of Fable's cells". */
  moves: string[];
}

export interface StandingEntry {
  handle: string;
  value: number;
  you: boolean;
  out: boolean;
  note?: string;
}

const MOVES_SHOWN = 10;

/** Leader first, the bot as "you", the last of three or more as "down to", anyone out at the end. */
export function standingLine(entries: StandingEntry[], unit: string): string {
  const live = entries.filter((e) => !e.out).sort((a, b) => b.value - a.value);
  const name = (e: StandingEntry) => (e.you ? 'you' : e.handle);
  const note = (e: StandingEntry) => (e.note ? ` (${e.note})` : '');
  const parts: string[] = [];
  if (live.length > 0) {
    const top = live.filter((e) => e.value === live[0].value);
    if (top.length > 1) parts.push(`${top.map(name).join(' and ')} are level at ${live[0].value} ${unit}`);
    else parts.push(`${name(live[0])} ${live[0].you ? 'lead' : 'leads'} with ${live[0].value} ${unit}${note(live[0])}`);
    const rest = live.slice(top.length);
    rest.forEach((e, k) => {
      const last = live.length >= 3 && k === rest.length - 1;
      parts.push(last ? `${name(e)} ${e.you ? 'are' : 'is'} down to ${e.value}${note(e)}` : `${name(e)} ${e.you ? 'hold' : 'has'} ${e.value}${note(e)}`);
    });
  }
  for (const e of entries.filter((x) => x.out)) parts.push(`${name(e)} ${e.you ? 'are' : 'is'} out`);
  return parts.join(', ');
}

type Seat = MatchEnvelope['match']['seats'][number];

function standingOf(env: MatchEnvelope): string | null {
  const v = (env.view ?? {}) as Record<string, any>;
  const entry = (s: Seat, value: number, out = false, note?: string): StandingEntry => ({ handle: s.identity.handle, value, you: s.seat === env.you, out, note });
  const seats = env.match.seats;
  switch (env.match.gameId) {
    case 'critical-mass':
      return standingLine(seats.map((s) => entry(s, v.counts?.[s.seat]?.cells ?? 0, (v.eliminated ?? []).includes(s.seat))), 'cells');
    case 'checkers':
      return standingLine(
        seats.map((s) => {
          const mine = ((v.pieces ?? []) as { side: string; king: boolean }[]).filter((p) => p.side === s.seat);
          const crowns = mine.filter((p) => p.king).length;
          return entry(s, mine.length, false, crowns > 0 ? `${crowns} crowned` : undefined);
        }),
        'knights',
      );
    case 'lightline':
      return standingLine(seats.map((s) => entry(s, v.score?.[s.seat] ?? 0)), 'rounds');
    default:
      return null;
  }
}

function turnOf(env: MatchEnvelope): string {
  const { match } = env;
  if (match.status === 'countdown') return 'The match is about to start';
  if (match.status !== 'live') return 'The match is over';
  if (env.realtime) return `Round ${(env.view as { round?: number } | null)?.round ?? 1}`;
  const n = (env.recent?.at(-1)?.number ?? 0) + 1;
  const mover = match.toMove[0];
  if (!mover) return `Turn ${n}`;
  if (mover === env.you) return `Turn ${n} — your move`;
  return `Turn ${n} — @${match.seats.find((s) => s.seat === mover)?.identity.handle ?? mover} to move`;
}

/** The table as a resident should see it before it opens its mouth. */
export function tableContext(env: MatchEnvelope): TableContext {
  return {
    players: env.match.seats.map((s) => `@${s.identity.handle} (${s.seat === env.you ? 'you' : s.identity.kind === 'human' ? 'human' : 'machine'})`).join(', '),
    turn: turnOf(env),
    standing: standingOf(env),
    moves: (env.recent ?? []).slice(-MOVES_SHOWN).map((r) => (r.handle ? `${r.number}. ${r.handle} ${r.text}` : `${r.number}. ${r.text}`)),
  };
}
```

```ts
/** A handle as people type it in passing: OLD_IRON → "old iron", "old_iron", "oldiron", any case;
 *  never part of a longer name (not "@OLD_IRON-2", not "sprockets"). */
export function callName(handle: string): RegExp {
  const parts = handle.split(/[\s_-]+/).filter(Boolean).map((p) => p.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'));
  return new RegExp(`(?<![A-Za-z0-9_-])${parts.join('[\\s_-]?')}(?![A-Za-z0-9_-])`, 'i');
}
```
